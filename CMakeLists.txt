# EmuSuNi: a Super Nintendo emulator for jailbroken PS4 consoles (Snes9x core), with the launcher,
# in-game menu and settings of Dolphin for PS4 (github.com/iHaiDeeZ/DolphinPS4).
#
# Configured by scripts/build.sh with toolchain/ps4-love-modern.cmake (clang 21 + libc++ 21 on the
# PacBrew OpenOrbis C runtime). Mesa's RADV (scripts/build-mesa.sh of DolphinPS4) draws everything.

project(emusuni C CXX ASM)

set(MESA_SRC_DIR $ENV{HOME}/dolphinps4-build/src/mesa CACHE PATH "Mesa source (Vulkan headers)")

# --- The emulator cores -------------------------------------------------------------------------
# Every core is a libretro core with the same API (retro_init, retro_run...). Several of them in
# one program: each core's API gets its own prefix (snes9x_retro_init, fceumm_retro_init), through
# a header forced into its sources; src/Cores.cpp collects them.
set(LIBRETRO_API retro_set_environment retro_set_video_refresh retro_set_audio_sample
  retro_set_audio_sample_batch retro_set_input_poll retro_set_input_state retro_init retro_deinit
  retro_api_version retro_get_system_info retro_get_system_av_info retro_set_controller_port_device
  retro_reset retro_run retro_serialize_size retro_serialize retro_unserialize retro_cheat_reset
  retro_cheat_set retro_load_game retro_load_game_special retro_unload_game retro_get_region
  retro_get_memory_data retro_get_memory_size)
function(core_prefix_header core)
  set(text "// Generated by CMakeLists.txt: the ${core} core's libretro API, renamed.
#pragma once
")
  foreach(fn ${LIBRETRO_API})
    string(APPEND text "#define ${fn} ${core}_${fn}
")
  endforeach()
  file(WRITE ${CMAKE_BINARY_DIR}/cores/prefix_${core}.h "${text}")
endfunction()
core_prefix_header(snes9x)
core_prefix_header(fceumm)

# --- The GPU filters' shaders (src/shaders: GLSL to SPIR-V headers for GpuFilter.cpp) -----------
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
file(GLOB EMUSUNI_SHADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/*.vert
  ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/*.frag ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/anime4k/*.frag)
set(EMUSUNI_SHADER_HEADERS)
foreach(shader ${EMUSUNI_SHADERS})
  get_filename_component(name ${shader} NAME)
  string(MAKE_C_IDENTIFIER ${name} id)
  set(header ${CMAKE_BINARY_DIR}/shaders/${name}.h)
  add_custom_command(OUTPUT ${header}
    COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -I${CMAKE_CURRENT_SOURCE_DIR}/src/shaders
            -I${CMAKE_CURRENT_SOURCE_DIR}/src/third_party/ffx --vn emusuni_spv_${id} -o ${header} ${shader}
    DEPENDS ${shader} ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/common.glsl
    COMMENT "Shader ${name}"
    VERBATIM)
  list(APPEND EMUSUNI_SHADER_HEADERS ${header})
endforeach()
add_custom_target(emusuni_shaders DEPENDS ${EMUSUNI_SHADER_HEADERS})
